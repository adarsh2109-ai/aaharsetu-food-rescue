    // Dedicated Donate Form Submission (User enters exact plates and items)
    function handleDedicatedDonate(e) {
      e.preventDefault();

      const sourceRadio = document.querySelector('input[name="pageSourceType"]:checked');
      if (!sourceRadio) {
        showToast('Please Select', 'Kripya khane ka source chunein (Wedding, Hotel, Home, ya Langar).');
        return;
      }

      const sourceType = sourceRadio.value;
      const dietary = document.getElementById('pDonateDietary').value;
      const servings = parseInt(document.getElementById('pDonateServings').value);
      const venue = document.getElementById('pDonateVenue').value.trim();
      const items = document.getElementById('pDonateItems').value.trim();
      const hoursLeft = parseInt(document.getElementById('pDonateHours').value);
      const packaging = document.getElementById('pDonatePackaging').value.trim();
      const city = document.getElementById('pDonateCity').value.trim();
      const phone = document.getElementById('pDonatePhone').value.trim();
      const address = document.getElementById('pDonateAddress').value.trim();

      if (!servings || servings <= 0) {
        showToast('Invalid Count', 'Kripya plates ki sahi sankhya daalein.');
        return;
      }

      const labels = { veg: 'Pure Veg', jain: 'Satvik / Jain', nonveg: 'Non-Veg' };
      const titles = {
        wedding: 'Wedding Function Surplus Food',
        hotel: 'Restaurant / Hotel Surplus Food',
        home: 'Ghar Ka Surplus Khana',
        temple: 'Prasad / Langar Surplus'
      };

      const newListing = {
        id: 'L-' + Date.now().toString().slice(-4),
        title: `${titles[sourceType] || 'Surplus Food'} (${servings} Plates)`,
        sourceType,
        venue,
        city,
        address,
        servings,
        dietary,
        dietaryLabel: labels[dietary] || 'Veg',
        items,
        packaging,
        hoursLeft,
        postedAgo: 'Just now',
        phone,
        status: 'available',
        claimedBy: null,
        claimedServings: null,
        riderName: null
      };

      listings.unshift(newListing);
      document.getElementById('dedicatedDonateForm').reset();
      updateBadges();

      showToast('Khana Post Ho Gaya! 🙏', `${servings} plates ki listing publish ho chuki hai.`);
      
      setTimeout(() => {
        navigateTo('explore');
        renderListings(listings);
      }, 700);
    }

    // Feed Filters
    function applyFilters() {
      const q = (document.getElementById('feedSearch').value || '').toLowerCase().trim();
      const cat = document.getElementById('feedCategory').value;
      const diet = document.getElementById('feedDietary').value;

      const filtered = listings.filter(item => {
        if (cat !== 'all' && item.sourceType !== cat) return false;
        if (diet !== 'all' && item.dietary !== diet) return false;
        if (q) {
          const matchTitle = item.title.toLowerCase().includes(q);
          const matchVenue = item.venue.toLowerCase().includes(q);
          const matchCity = item.city.toLowerCase().includes(q);
          const matchItems = item.items.toLowerCase().includes(q);
          if (!matchTitle && !matchVenue && !matchCity && !matchItems) return false;
        }
        return true;
      });

      renderListings(filtered);
    }

    async function handleAiChatSubmit(e) {
      if (e) e.preventDefault();
      const input = document.getElementById('aiChatInput');
      const query = input.value.trim();
      if (!query) return;

      input.value = '';
      appendChatMessage('user', query);

      // Loading bubble
      const loadingId = 'ai-loading-' + Date.now();
      appendChatLoading(loadingId);

      const systemPrompt = `You are Setu AI, a friendly, empathetic Hindi/Hinglish speaking assistant for the surplus food rescue platform 'AaharSetu' (आहार सेतु). 
Your job is to guide users on:
1. Donating leftover food from weddings, parties, restaurants, and homes.
2. Safety and packaging guidelines (keep dry & liquid foods separate, wrap within clean foil, safe within 4 hours).
3. How delivery partners/riders pick up and deliver food to nearby shelters and needy individuals.
4. Answering questions concisely, politely, and practically in conversational Hindi/Hinglish. Keep answers short and direct (2-4 sentences max).`;

      try {
        const apiKey = "";
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

        const payload = {
          contents: [{ parts: [{ text: query }] }],
          systemInstruction: { parts: [{ text: systemPrompt }] }
        };

        const response = await fetch(apiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        const data = await response.json();
        removeChatLoading(loadingId);

        if (data.candidates && data.candidates[0]?.content?.parts?.[0]?.text) {
          const reply = data.candidates[0].content.parts[0].text;
          appendChatMessage('bot', reply);
        } else {
          appendChatMessage('bot', 'Shukriya! Aap donate form ya rider portal use karke seedha khana deliver kar sakte hain.');
        }
      } catch (err) {
        removeChatLoading(loadingId);
        appendChatMessage('bot', 'Aapka sawal sun liya! Surplus khana jaldi se safe bartan me pack karein aur "Post Khana" se details share karein taaki rider pick kar sake.');
      }
    }

    function appendChatMessage(sender, text) {
      const container = document.getElementById('aiChatMessages');
      const msgDiv = document.createElement('div');
      msgDiv.className = `flex gap-2 ${sender === 'user' ? 'justify-end' : ''}`;

      if (sender === 'user') {
        msgDiv.innerHTML = `
          <div class="bg-purple-600 text-white p-3 rounded-2xl rounded-tr-none text-xs leading-relaxed max-w-[80%] shadow-sm">
            ${text}

            ${text}
          </div>
        `;

    }
